        <p class="instruction">Sviluppa l'armonia dei tre centri</p>

        <div class="centers-container">
            <div id="mente" class="center-node mente" onclick="toggleCenter('mente')"><span>MENTE</span></div>
            <div id="cuore" class="center-node cuore" onclick="toggleCenter('cuore')"><span>CUORE</span></div>
            <div id="pancia" class="center-node pancia" onclick="toggleCenter('pancia')"><span>PANCIA</span></div>
        </div>

        <div id="harmony-msg">La tua coscienza illumina l'umanità</div>
    </div>

    <script>
        const canvas = document.getElementById('starsCanvas');
        const ctx = canvas.getContext('2d');
        let width, height;
        let particles = [];
        let trails = [];
        let state = { mente: false, cuore: false, pancia: false };
        let harmony = false;

        // Inizializzazione Canvas
        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resize);
        resize();

        // Classe Particella (Umani-Stelle)
        class Particle {
            constructor() {
                this.reset();
            }
            reset() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = Math.random() * 2;
                this.speedX = (Math.random() - 0.5) * 0.5;
                this.speedY = (Math.random() - 0.5) * 0.5;
                this.alpha = Math.random();
            }
            update(mouseX, mouseY) {
                if (harmony) {
                    // Legge del Destino Comune: seguono la luce guida
                    let dx = mouseX - this.x;
                    let dy = mouseY - this.y;
                    let dist = Math.sqrt(dx*dx + dy*dy);
                    if (dist < 300) {
                        this.x += dx * 0.02;
                        this.y += dy * 0.02;
                    }
                    this.alpha = 1;
                } else {
                    this.x += this.speedX;
                    this.y += this.speedY;
                    if (this.x < 0 || this.x > width) this.speedX *= -1;
                    if (this.y < 0 || this.y > height) this.speedY *= -1;
                }
            }
            draw() {
                ctx.fillStyle = `rgba(255, 255, 255, ${this.alpha})`;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        // Creazione stelle
        for (let i = 0; i < 150; i++) {
            particles.push(new Particle());
        }

        let mouse = { x: -1000, y: -1000 };
        window.addEventListener('mousemove', e => {
            mouse.x = e.clientX;
            mouse.y = e.clientY;
            if (harmony) {
                // Crea la scia (Modello di riferimento)
                trails.push({ x: mouse.x, y: mouse.y, life: 100 });
            }
        });

        function toggleCenter(type) {
            state[type] = !state[type];
            document.getElementById(type).classList.toggle(`active-${type}`);
            
            // Feedback vibrazione se disponibile
            if ("vibrate" in navigator) navigator.vibrate(50);

            checkHarmony();
        }

        function checkHarmony() {
            if (state.mente && state.cuore && state.pancia) {
                harmony = true;
                document.getElementById('harmony-msg').style.opacity = "1";
            } else {
                harmony = false;
                document.getElementById('harmony-msg').style.opacity = "0";
            }
        }

        function animate() {
            // Effetto scia per il background
            ctx.fillStyle = `rgba(5, 5, 8, ${harmony ? 0.1 : 0.3})`;
            ctx.fillRect(0, 0, width, height);

            // Disegna le scie del leader
            trails.forEach((t, i) => {
                ctx.beginPath();
                ctx.arc(t.x, t.y, t.life / 10, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(255, 255, 255, ${t.life / 100})`;
                ctx.fill();
                t.life -= 2;
                if (t.life <= 0) trails.splice(i, 1);
            });

            particles.forEach(p => {
                p.update(mouse.x, mouse.y);
                p.draw();
            });

            requestAnimationFrame(animate);
        }

        animate();
    </script>
